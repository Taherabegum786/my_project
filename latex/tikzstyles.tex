% 3-D slab: \pic at (x,y) {slab={width}{height}{depth}{colour}};  front-lower-left corner at the pic origin.
\tikzset{
  pics/slab/.style n args={4}{code={
    \fill[#4] (0,0) rectangle (#1,#2);
    \fill[#4!60!white] (0,#2) -- ++({0.55*#3},{0.35*#3}) -- ++(#1,0) -- ++({-0.55*#3},{-0.35*#3}) -- cycle;
    \fill[#4!78!black] (#1,0) -- ++({0.55*#3},{0.35*#3}) -- ++(0,#2) -- ++({-0.55*#3},{-0.35*#3}) -- cycle;
    \draw[fgink, line width=0.5pt] (0,0) rectangle (#1,#2)
      (0,#2) -- ++({0.55*#3},{0.35*#3}) -- ++(#1,0) -- ++({-0.55*#3},{-0.35*#3})
      (#1,0) -- ++({0.55*#3},{0.35*#3}) -- ++(0,#2);
  }},
  % 3-D disk / cylinder seen slightly from above
  pics/disk/.style n args={3}{code={% width, height, colour
    \fill[#3!85!black] (0,0) arc[start angle=180, end angle=360, x radius={#1/2}, y radius={#1/7}]
      -- ++(0,#2) arc[start angle=0, end angle=-180, x radius={#1/2}, y radius={#1/7}] -- cycle;
    \shade[left color=#3!70!black, right color=#3!70!black, middle color=#3!35!white]
      (0,0) arc[start angle=180, end angle=360, x radius={#1/2}, y radius={#1/7}]
      -- ++(0,#2) arc[start angle=0, end angle=-180, x radius={#1/2}, y radius={#1/7}] -- cycle;
    \fill[#3!40!white] ({#1/2},#2) ellipse[x radius={#1/2}, y radius={#1/7}];
    \draw[fgink, line width=0.5pt] ({#1/2},#2) ellipse[x radius={#1/2}, y radius={#1/7}]
      (0,#2) -- (0,0) arc[start angle=180, end angle=360, x radius={#1/2}, y radius={#1/7}] -- (#1,#2);
  }},
}

% Ornamental rule used in chapter openings and boxes
\newcommand{\ornament}{\tikz[baseline=-0.6ex]{\draw[fwine, line width=0.4pt] (0,0) -- (1.6,0);
  \fill[fwine] (1.85,0) circle[radius=0.75pt]; \fill[fwine] (2.05,0) circle[radius=1.1pt]; \fill[fwine] (2.25,0) circle[radius=0.75pt];
  \draw[fwine, line width=0.4pt] (2.5,0) -- (4.1,0);}}

% Gantt chart row: \ganttrow{cm per time unit}{y}{name/start/end/colour, ...}
\newcommand{\ganttrow}[3]{%
  \foreach \gtP/\gtS/\gtE/\gtC in {#3} {
    \filldraw[fill=\gtC, draw=fgink, line width=0.5pt] ({\gtS*#1},#2) rectangle ({\gtE*#1},{#2+0.75});
    \node[font=\small] at ({(\gtS+\gtE)/2*#1},{#2+0.375}) {\gtP};
    \node[font=\scriptsize, anchor=north] at ({\gtS*#1},{#2-0.03}) {\gtS};
    \node[font=\scriptsize, anchor=north] at ({\gtE*#1},{#2-0.03}) {\gtE};
  }}
