\begin{formulatable}
Differences and number of steps & \Delta x = x_2 - x_1,\quad \Delta y = y_2 - y_1,\quad \text{steps} = \max(|\Delta x|, |\Delta y|) \\
Increments & x_{inc} = \frac{\Delta x}{\text{steps}},\qquad y_{inc} = \frac{\Delta y}{\text{steps}} \\
Each step & x \leftarrow x + x_{inc},\ \ y \leftarrow y + y_{inc},\ \ \text{plot}(\operatorname{round}(x), \operatorname{round}(y)) \\
\end{formulatable}
