\begin{formulatable}
Union & \mu_{A \cup B}(x) = \max(\mu_A(x), \mu_B(x)) \\
Intersection & \mu_{A \cap B}(x) = \min(\mu_A(x), \mu_B(x)) \\
Complement & \mu_{A'}(x) = 1 - \mu_A(x) \\
Algebraic sum; algebraic product & \mu_A + \mu_B - \mu_A\mu_B;\quad \mu_A\,\mu_B \\
Bounded sum; bounded difference & \min(1, \mu_A + \mu_B);\quad \max(0, \mu_A - \mu_B) \\
Concentration (``very''); dilation (``somewhat'') & \mu^2;\quad \sqrt{\mu} \\
\end{formulatable}
