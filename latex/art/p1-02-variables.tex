  \node[slate, text width=30mm] (IV) {\textbf{Independent}\\\footnotesize cause — manipulated};
  \node[grey, text width=30mm, right=of IV] (MV) {\textbf{Mediating}\\\footnotesize the mechanism};
  \node[sage, text width=30mm, right=of MV] (DV) {\textbf{Dependent}\\\footnotesize effect — measured};
  \node[sand, text width=30mm, above=10mm of MV] (MOD) {\textbf{Moderator}\\\footnotesize alters the strength};
  \node[rose, text width=34mm, below=10mm of DV] (EX) {\textbf{Extraneous}\\\footnotesize must be controlled};
  \draw[arr] (IV) -- (MV); \draw[arr] (MV) -- (DV);
  \draw[dasharr] (MOD) -- ($(MV.north)!0.5!(DV.north)+(0,0)$);
  \draw[dasharr] (EX) -- (DV);
\end{tikzpicture}
\caption{How variables relate in a study. Example: teaching method (independent) raises motivation
(mediating), which raises achievement (dependent); the effect may be stronger for younger pupils
(moderator) and can be distorted by prior knowledge (extraneous).}
