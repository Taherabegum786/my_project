\toprule
\textbf{Form} & \textbf{Premises} & \textbf{Conclusion} & \textbf{Valid?} \\
\midrule
Modus ponens & $P \rightarrow Q$,\ \ $P$ & $Q$ & yes \\
Modus tollens & $P \rightarrow Q$,\ \ $\neg Q$ & $\neg P$ & yes \\
Affirming the consequent & $P \rightarrow Q$,\ \ $Q$ & $P$ & \textbf{no} \\
Denying the antecedent & $P \rightarrow Q$,\ \ $\neg P$ & $\neg Q$ & \textbf{no} \\
\bottomrule
\end{tabular}

\smallskip
{\small ``$P$ only if $Q$'' means $P \rightarrow Q$;\quad ``$P$ unless $Q$'' means $\neg Q \rightarrow P$.}
