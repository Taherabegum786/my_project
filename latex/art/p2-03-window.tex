\begin{formulatable}

Vertical mapping & y_v = y_{v\min} + (y_w - y_{w\min})\,s_y,\quad s_y = \frac{y_{v\max}-y_{v\min}}{y_{w\max}-y_{w\min}} \\
\end{formulatable}
