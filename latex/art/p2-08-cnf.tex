\begin{enumerate}
\item Replace terminals in bodies of length $\ge 2$ by new variables: $A \to a$, $B \to b$, giving
$S \to A\,S\,B \mid A\,B$.
\item Break bodies longer than two symbols: $S \to A\,X$ with $X \to S\,B$.
\end{enumerate}
\[
\textbf{CNF:}\quad S \to AX \mid AB, \qquad X \to SB, \qquad A \to a, \qquad B \to b
\]
