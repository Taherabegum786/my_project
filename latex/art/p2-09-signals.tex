\begin{formulatable}
Nyquist bit rate (noiseless channel, bandwidth $B$, $L$ signal levels) & \text{bit rate} = 2B\log_2 L \\
Shannon capacity (noisy channel) & C = B\log_2(1 + \text{SNR}) \\
Signal-to-noise ratio in decibels & \text{SNR}_{\text{dB}} = 10\log_{10}\text{SNR}\quad (30\text{ dB} = 1000) \\
Bandwidth--delay product (bits ``in the pipe'') & \text{bandwidth} \times \text{propagation delay} \\
Propagation and transmission delay & T_p = \frac{\text{distance}}{\text{speed}},\quad T_t = \frac{\text{frame size}}{\text{bandwidth}} \\
Latency & T_p + T_t + \text{queuing} + \text{processing} \\
Baud (signal) rate, $r$ bits per signal element & S = N / r \\
\end{formulatable}
