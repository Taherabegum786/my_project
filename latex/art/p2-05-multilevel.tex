\caption{A 48-bit virtual address with 4 KB pages and 8-byte page-table entries: four 9-bit indexes and a
12-bit offset (the x86-64 layout).}

\begin{formulas}
\begin{align*}
\text{offset} &= \log_2 4096 = 12 \text{ bits} &\Rightarrow\ \text{page-number bits} &= 48 - 12 = 36\\
\text{entries per table page} &= 4096 / 8 = 512 = 2^9 &\Rightarrow\ \text{bits per level} &= 9\\
\text{levels} &= \lceil 36/9 \rceil = 4 &
\text{accesses without TLB} &= 4 + 1 = 5
\end{align*}
\end{formulas}
