\begin{formulas}
\begin{align*}
\text{translate by }(-2,-2):\quad &(4,2)\mapsto(2,0)\\
\text{rotate by }90^\circ:\quad &(x\cos 90^\circ - y\sin 90^\circ,\ x\sin 90^\circ + y\cos 90^\circ) = (0,2)\\
\text{translate by }(+2,+2):\quad &(0,2)\mapsto(2,4) = P'
\end{align*}
\end{formulas}
