\begin{formulatable}
Ratio of propagation to transmission time & a = T_p / T_t \\
Stop-and-wait efficiency & \eta = \frac{1}{1 + 2a} \\
Sliding window of size $W$ & \eta = \frac{W}{1 + 2a}\ \ \text{if } W < 1 + 2a,\ \text{else } 1 \\
Window for full utilisation & W \ge 1 + 2a \\
Go-Back-N with $n$-bit sequence numbers & W_s \le 2^n - 1,\quad W_r = 1 \\
Selective Repeat with $n$-bit sequence numbers & W_s = W_r \le 2^{n-1} \\
Throughput & \eta \times \text{bandwidth} \\
\end{formulatable}
