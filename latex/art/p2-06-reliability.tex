\begin{formulatable}
Mean time between failures & \text{MTBF} = \text{MTTF} + \text{MTTR} \\
Availability & A = \frac{\text{MTTF}}{\text{MTTF} + \text{MTTR}} \times 100\% \\
Reliability with constant failure rate $\lambda$ & R(t) = e^{-\lambda t} \\
Rate of occurrence of failure (ROCOF) & \text{failures per unit time} \\
Probability of failure on demand (POFOD) & \Pr(\text{failure} \mid \text{demand}) \\
\end{formulatable}
